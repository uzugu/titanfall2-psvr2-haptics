cmake_minimum_required(VERSION 3.20)
project(Titanfall2VR_PSVR2 VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(effect_core STATIC src/effects.cpp)
target_include_directories(effect_core PUBLIC src)
if(WIN32)
  add_library(Titanfall2VR_PSVR2 SHARED src/plugin.cpp src/runtime.cpp)
  target_link_libraries(Titanfall2VR_PSVR2 PRIVATE effect_core)
  target_compile_definitions(Titanfall2VR_PSVR2 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_executable(psvr2_probe src/probe.cpp)
  add_executable(plugin_smoke tests/plugin_smoke.cpp)
  add_library(mock_capi SHARED tests/mock_capi.cpp)
  target_link_libraries(mock_capi PRIVATE effect_core)
  set_target_properties(mock_capi PROPERTIES OUTPUT_NAME psvr2_toolkit_capi RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mock")
  add_executable(runtime_test tests/runtime_test.cpp src/runtime.cpp)
  target_link_libraries(runtime_test PRIVATE effect_core)
  target_compile_definitions(runtime_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_target_properties(runtime_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/harness")
endif()
enable_testing()
add_executable(effect_tests tests/effects_test.cpp)
target_link_libraries(effect_tests PRIVATE effect_core)
add_test(NAME effect_tests COMMAND effect_tests)
find_program(NODE_EXECUTABLE node)
if(NODE_EXECUTABLE)
  add_test(NAME catalog_test COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/catalog.test.mjs")
endif()
if(WIN32)
  add_test(NAME plugin_smoke COMMAND plugin_smoke $<TARGET_FILE:Titanfall2VR_PSVR2>)
  add_test(NAME runtime_test COMMAND runtime_test $<TARGET_FILE:mock_capi>)
endif()
if(MSVC)
  foreach(target effect_core effect_tests Titanfall2VR_PSVR2)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
  endforeach()
endif()
